\section{Temporal reconstruction}
\begin{frame}{Start with one pixel: integrated changes on a temporal chain}
\begin{columns}[T]
\column{.54\textwidth}
Choose a unit-mass temporal kernel $\phi_{s_k}$:
\[e(t)=\sum_{k\text{ at this pixel}}p_k\phi_{s_k}(t).\]
\[y_i=\int_{t_i}^{t_{i+1}}e(t)\,dt.\]
The reconstruction times $t_i$ are chosen by the algorithm.
\column{.43\textwidth}
\[\A=\begin{pmatrix}-1&1&&\\&-1&1&\\&&\ddots&\ddots\end{pmatrix}.\]
\[\A\mathbf f\approx\mathbf y.\]
\[\ker\A=\operatorname{span}\{\mathbf1\}.\]
Events determine increments, leaving a constant free.
\end{columns}
\vspace{1mm}
{\footnotesize\textbf{In scene terms:} each pixel's log-brightness history is rebuilt on its own from how much it rose or fell in each interval. The model chooses the reference level; a pixel without events stays at that level, whatever its brightness.\par}
\takeaway{Kernel width controls temporal averaging; a baseline requires an additional choice.}
% source detail: Ch. 2, eq:exy, eq:Exy, eq:tempdiffA, lem:nullspace.
\src{\dref{Eq.}{eq:Exy}, \dref{Eq.}{eq:tempdiffA}, and \dref{Lemma}{lem:nullspace}.}
\talknote{Temporal observation model; 75 seconds.}{Suppress the pixel index to keep the notation readable. Integrating a unit-mass kernel across a bin assigns all or part of each event to that bin. Delta binning has no spread; Gaussian, rectangular, and spline kernels distribute influence over time. The entries of A are unscaled increments, not derivatives divided by a time step.}{If Af=0, every consecutive pair of components is equal, hence f is constant. Conversely A1=0. Under ideal threshold calibration the observed increments approximate changes in L/c\_th, with kernel-boundary, overshoot, and between-event endpoint errors. No absolute frame is used to set the baseline. Scene terms: each pixel's brightness history is rebuilt on its own; a pixel without events stays at the reference level.}
\end{frame}

\begin{frame}{A regularized estimate is uniquely selected}
\small
For fixed $\Lam=\operatorname{diag}(\lambda_i)$ with every $\lambda_i>0$,
\[\mathbf f^*=\arg\min_{\mathbf f\in\R^n}\left\{\frac12\norm{\A\mathbf f-\mathbf y}^2+\frac12\norm{\Lam\mathbf f}^2\right\}.\]
The unique minimizer solves the normal equations:
\[\K_{\rm time}=\A^\top\A+\Lam^\top\Lam\succ0,\qquad \K_{\rm time}\mathbf f^*=\A^\top\mathbf y.\]
\textbf{Compute each pixel, then assemble frames}\par
\begin{enumerate}\footnotesize\setlength{\itemsep}{0pt}
% source: monocular_reconstruction/src/core/lat_image_af_with_lambda.m:87--96 and lat_image_af.m:52--61; independent pixel systems use MATLAB left division in parfor, then assemble L.
\item Form the pixel's matrix and right-hand side; solve the linear system numerically.
\item Run independent pixel solves in parallel; assemble their temporal profiles into frames.
\end{enumerate}
% source: Direct consequence of Ch. 2 thm:uniqueness normal equations and A1=0; supplementary identity.
The selected baseline satisfies $\mathbf1^\top\Lam^2\mathbf f^*=0$: intervals with larger weights set the reference level.

% source detail: Ch. 2, eq:in, thm:uniqueness; monocular reconstruction implementation, per-pixel linear solves.

\begin{frame}{Why uniqueness and stability hold}
Write $\lambda_{\min}=\min_i\lambda_i$. For $\mathbf z\ne0$,
\[\mathbf z^\top\K_{\rm time}\mathbf z=\norm{\A\mathbf z}^2+\norm{\Lam\mathbf z}^2\ge\lambda_{\min}^2\norm{\mathbf z}^2>0.\]
At the same fixed weights,
\[\delta\mathbf f^*=\K_{\rm time}^{-1}\A^\top\delta\mathbf y,\qquad
\norm{\delta\mathbf f^*}\le\frac{\norm{\A}_2}{\lambda_{\min}^2}\norm{\delta\mathbf y}.\]
% source: Ch. 2 thm:uniqueness and prop:stability; analytic mixed-norm bound ||A||_2 <= 2.
\[\norm{\A}_2\le2,\qquad \kappa_2(\K_{\rm time})\le\frac{4+\lambda_{\max}^2}{\lambda_{\min}^2}.\]
% source: nonnegative unit-mass kernels split one event's polarity across bins, so ||delta y||_2 <= ||delta y||_1 = 1 (Ch. 2 prop:l1_bump_properties).
{\small One spurious or missed event gives $\norm{\delta\mathbf y}\le1$: the bound limits how far one noise event moves a pixel's history.\par}
\takeaway{Coercivity gives a stable estimate; weakening the penalty weakens these full-space bounds.}
% source detail: Ch. 2, thm:uniqueness, prop:stability.
\src{\dref{Theorem}{thm:uniqueness}, \dref{Proposition}{prop:stability}.}
% source: direct sharper operator bound from Ch. 2 normal equations and Lambda^2 >= lambda_min^2 I.
\talknote{Proof; 90 seconds.}{Walk through the quadratic form. Both terms are nonnegative, and the strictly positive diagonal penalty supplies a lower bound in every direction. Finite-dimensional coercivity gives existence; the positive-definite Hessian gives uniqueness. Subtracting the two normal equations gives the exact perturbation identity.}{The displayed full-space bound uses $\norm{\K^{-1}}_2\le\lambda_{\min}^{-2}$. A sharper operator bound is $\norm{\K^{-1}\A^\top}_2\le1/(2\lambda_{\min})$: from $\A^\top\A\preceq\K-\lambda_{\min}^2I$, the squared singular-value bound is at most $\max_{t\ge\lambda_{\min}^2}(1/t-\lambda_{\min}^2/t^2)=1/(4\lambda_{\min}^2)$. The displayed estimate is a worst-case bound; the reported condition numbers describe the evaluated configurations. Scene terms: one spurious or missed event gives $\norm{\delta\mathbf y}\le1$ (nonnegative unit-mass kernels), so the bound limits how far one noise event moves a history.}
\end{frame}

\begin{frame}{Derive the time-surface constraint by following the edge}
\small
For an ideal single crossing, define
\[\psi(\mathbf x,t)=t-\mathcal T(\mathbf x),\qquad \psi(\mathbf x(t),t)=0.\]
If the edge translates with constant $\dot{\mathbf x}=\vct$, then
\[0=\frac d{dt}\psi(\mathbf x(t),t)=1-\nabla\mathcal T\cdot\vct.\]
{\footnotesize In scene terms: $|\nabla\mathcal T|$ is edge slowness, normal speed $1/|\nabla\mathcal T|$; motion along the edge leaves $\mathcal T$ fixed.\par}
\begin{block}{Model hypotheses}
The edge crosses each location once, has nonzero normal speed, and its recorded timestamp equals a differentiable crossing-time map.
\end{block}
\takeaway{Interpolation, stale timestamps, and repeated crossings make this an approximate data constraint.}
% source detail: Ch. 3, prop:time\_surface; def:time\_surface.
\src{\dref{Proposition}{prop:time_surface}; \dref{Definition}{def:time_surface}; time surfaces~\cite{benosman2012asynchronous,benosman2014event}.}
\talknote{Geometric derivation; 60 seconds.}{This chain-rule argument is accessible without any sensor-specific machinery. Follow a material point on the translating edge and differentiate the identity saying it stays on the zero level set.}{The coefficient bounds used later apply to fitted or mollified maps. The timestamp-regression result assumes a differentiable crossing-time map with Lipschitz Hessian, but raw last-timestamp maps do not automatically satisfy those assumptions. A timestamp may belong to an old edge or multiple crossings; the variational theorem bounds usable coefficient perturbations under uniform identifiability. Scene terms: $|\nabla\mathcal T|$ is slowness; an edge taking 10 ms per pixel moves at 100 px/s normal to itself.}
\end{frame}

\begin{frame}{Voxel projection: transport a represented event-activity field}
Smooth and bin events into a spatiotemporal field $V(\mathbf x,t)$; project over a window, $a(\mathbf x,s)=\int_s^{s+T}V(\mathbf x,t)\,dt$.
\begin{block}{Modeling assumption (added to the threshold model)}
The activity field moves with the scene: $V_t+\vct\cdot\nabla V=0$, with constant $\vct$ over the window.
\end{block}
Integrating over the window then gives
\[a_s+\vct\cdot\nabla a=0,\qquad a_s=V(\mathbf x,s+T)-V(\mathbf x,s),\]
so the voxel-projection residual is $\nabla a\cdot\vct+a_s$.
\takeaway{The threshold model supplies the activity field; transporting it with the scene is a separate modeling choice.}
% source detail: Ch. 3, subsec:vox\_projection; derivative-operator qualifications.
\src{\dref{Section}{subsec:vox_projection}; derivative-operator qualifications; voxel grid~\cite{zhu2019unsupervised}.}
\talknote{Voxel model; 75 seconds.}{A voxel grid adds a time axis to the image plane and stores represented event contributions in time bins. The activity field is not an intensity image: a bright stationary scene can have zero activity. The transport equation is an assumption about this represented activity; the threshold model does not imply it, because a pixel can start or stop producing events as brightness, motion, and threshold conditions change. For an exactly transported density with constant velocity, integrate the PDE over the sliding window; this gives the displayed transport identity exactly, even for a large displacement.}{Finite temporal projection changes the profile by averaging its translates. The evaluated spatial bin mean and temporal OLS slope need not be derivative-consistent partners, and asymmetric smoothing can add mismatch. The well-posedness theorem only needs fixed bounded coefficient fields; an accurate physical-velocity interpretation also needs the transport approximation.}
\end{frame}

\begin{frame}{The structure tensor pools directional information}
At a fixed evaluation time $t_\star$, set
\[\mathbf J(\mathbf x)=\int W(\mathbf x-\mathbf y)\,\nabla_3V(\mathbf y,t_\star)\nabla_3V(\mathbf y,t_\star)^\top\,d\mathbf y,
\qquad W\ge0.\]
Use the local residual energy
\[\vct_3^\top\mathbf J\vct_3,\qquad \vct_3=(v_x,v_y,1)^\top,\qquad
\mathbf J=\begin{pmatrix}\widetilde{\mathbf J}&\mathbf j_t\\\mathbf j_t^\top&J_{tt}\end{pmatrix}.\]
\begin{itemize}
\item Each gradient outer product is positive semidefinite and rank at most one.
\item Pooling nonparallel spatial gradients can make $\widetilde{\mathbf J}$ full rank.
\end{itemize}
\takeaway{Pooling can identify both components locally; the global theorem needs only integrated identifiability.}
% source detail: Ch. 3, subsec:vox\_struct, prop:vox\_tensor\_wellposed.
\src{\dref{Section}{subsec:vox_struct}, \dref{Proposition}{prop:vox_tensor_wellposed}; spatiotemporal-surface view~\cite{benosman2012asynchronous}.}
\talknote{Structure tensor; 75 seconds.}{Expand the quadratic: v\textasciicircum{}T Jtilde v + 2 j\_t\textasciicircum{}T v + J\_tt. The last term is independent of velocity, so the spatial block is the data curvature and the temporal cross terms give the forcing. The temporal component of v\_3 is fixed to one; this is an ordinary least-squares transport model, not a total-least-squares eigenvector model.}{Nonnegative pooling preserves positive semidefiniteness, since z\textasciicircum{}T J z is an integral of nonnegative squared directional derivatives. A locally planar event surface is a model assumption, and smoothing scale limits temporal detail. The theorem will not require every pixel's spatial block to be positive definite.}
\end{frame}

\begin{frame}{The main flow theorem: global information controls constants}
\begin{block}{Unified well-posedness theorem}
Let $\Omega\subset\R^2$ be bounded, connected, and Lipschitz; $\alpha>0$.
Assume
\[\B\in L^\infty,\quad \B=\B^\top\succeq0\text{ a.e.},\quad b\in L^2,\qquad
\int_\Omega\B(\mathbf x)\,d\mathbf x\succ0.\]
Then $E_{\B,b}$ has a unique minimizer in $H^1(\Omega;\R^2)$.
\end{block}
\vspace{1mm}
The integrated matrix must identify \textbf{every constant velocity direction}.\par
\small In scene terms: edges in at least two orientations somewhere in the image; neighbors supply the motion where $\B=0$.
\takeaway{$\B$ may vanish on open regions; integrated directional information and smoothness still give a unique estimate.}
% source detail: Ch. 3, thm:unified\_wellposedness, eq:integrated\_identifiability.
\src{\dref{Theorem}{thm:unified_wellposedness}, \dref{Eq.}{eq:integrated_identifiability}; classical quadratic-flow theory~\cite{weickert2001theoretical}.}
\talknote{Central theorem; 90 seconds.}{Classical quadratic optical flow already has well-posedness theory, including Weickert and Schn\"orr. The contribution here is one analysis for three event-derived coefficient constructions, with explicit identifiability and perturbation hypotheses. Read the hypotheses and interpret each. Connectedness makes the kernel of the gradient consist of global constants. The positive regularization coefficient controls spatial variation. Bounded data curvature and L2 forcing make the weak forms bounded. Positive definiteness of the integrated B controls constant fields that smoothness cannot see.}{B may vanish on open subsets while its integral remains positive definite. For time surfaces, no nonzero vector can be perpendicular to all measured gradients almost everywhere. The condition need not hold pointwise. Scene terms: edges in at least two orientations somewhere in the image.}
\end{frame}

\begin{frame}{Proof idea: smoothness fixes variation, edges fix the drift}

{\footnotesize Poincar\'e~\cite{evans2010partial}}

\[m=\lambda_{\min}\Bigl(\int_\Omega\B\,d\mathbf x\Bigr)>0\]
{\footnotesize edges cover every direction}

\[\mathcal B(\vct,\vct)\ge c_0\norm{\vct}_{H^1}^2\]
{\footnotesize $c_0=c_0(\Omega,\alpha,\norm{\B}_{L^\infty},m)$}

\begin{frame}{Start with one pixel: integrated changes on a temporal chain}
\begin{columns}[T]
\column{.54\textwidth}
Choose a unit-mass temporal kernel $\phi_{s_k}$:
\[e(t)=\sum_{k\text{ at this pixel}}p_k\phi_{s_k}(t).\]
\[y_i=\int_{t_i}^{t_{i+1}}e(t)\,dt.\]
The reconstruction times $t_i$ are chosen by the algorithm.
\column{.43\textwidth}
\[\A=\begin{pmatrix}-1&1&&\\&-1&1&\\&&\ddots&\ddots\end{pmatrix}.\]
\[\A\mathbf f\approx\mathbf y.\]
\[\ker\A=\operatorname{span}\{\mathbf1\}.\]
Events determine increments, leaving a constant free.
\end{columns}

\begin{frame}{Continuous time: a distributional initial-value problem}
\small Assume $s_k\ne t_i$ for all $k,i$; set $\phi_{s,h}(t)=h^{-1}\phi_0((t-s)/h)$.
\[g_h(t)=f_0+\int_{t_0}^{t}\sum_k p_k\phi_{s_k,h}\,d\xi\ \xrightarrow[h\to0]{L^1}\ g(t)=f_0+\sum_{t_0<s_k\le t}p_k,\]
\[\frac{dg}{dt}=\mu:=\sum_k p_k\,\delta_{s_k}\ \text{ in }\mathcal D'((t_0,t_N)),\quad g(t_0^+)=f_0,\quad g(t_i)-g(t_{i-1})=\mu([t_{i-1},t_i)).\]
\begin{itemize}
\item $\A\mathbf f\approx\mathbf y$ discretizes this equation by finite differences.
\item $g\in BV(t_0,t_N)$, and $g\in H^1$ only when $\mu$ has no nonzero atom.
\item The events fix every jump of $g$; the initial value $f_0$ stays free.
\end{itemize}
\takeaway{The free initial value is the continuous form of $\ker\A=\operatorname{span}\{\mathbf1\}$. The regularizer supplies it.}
% source detail: Ch. 2, subsubsec:ode, eq:difference\_quotient, eq:ivp, eq:ivp\_integrated.
\src{\dref{Eq.}{eq:ivp} and \dref{Eq.}{eq:ivp_integrated}; continuous-time formulation (\drefpage{subsubsec:ode}).}
\talknote{Continuous-time connection; 90 seconds.}{Mollify each event at scale h; the integrated signal is a smooth staircase that converges in L1 to a pure jump function as h goes to zero. Its distributional derivative is the signed event measure, so the cumulative signal solves a measure-valued initial-value problem. Integrating over a reconstruction bin recovers exactly the observed net polarity, which is what the rows of A f approximately match.}{The cumulative signal g is the unregularized event integral: it is neither the physical log-intensity, which needs an absolute reference, nor the Tikhonov minimizer. A jump function with a nonzero jump is in BV but not in H1, which is why the method works with integrated bins rather than pointwise derivatives of raw events. The measure-valued limit is also on the discussion page on the jump primitive. Transition: since the data fix only increments, a penalty must choose the level, which is the next theorem.}
\end{frame}

\begin{frame}{A regularized estimate is uniquely selected}
\small
For fixed $\Lam=\operatorname{diag}(\lambda_i)$ with every $\lambda_i>0$,
\[\mathbf f^*=\arg\min_{\mathbf f\in\R^n}\left\{\frac12\norm{\A\mathbf f-\mathbf y}^2+\frac12\norm{\Lam\mathbf f}^2\right\}.\]
The unique minimizer solves the normal equations:
\[\K_{\rm time}=\A^\top\A+\Lam^\top\Lam\succ0,\qquad \K_{\rm time}\mathbf f^*=\A^\top\mathbf y.\]

\begin{frame}{Why uniqueness and stability hold}
Write $\lambda_{\min}=\min_i\lambda_i$. For $\mathbf z\ne0$,
\[\mathbf z^\top\K_{\rm time}\mathbf z=\norm{\A\mathbf z}^2+\norm{\Lam\mathbf z}^2\ge\lambda_{\min}^2\norm{\mathbf z}^2>0.\]
At the same fixed weights,
\[\delta\mathbf f^*=\K_{\rm time}^{-1}\A^\top\delta\mathbf y,\qquad
\norm{\delta\mathbf f^*}\le\frac{\norm{\A}_2}{\lambda_{\min}^2}\norm{\delta\mathbf y}.\]
% source: Ch. 2 thm:uniqueness and prop:stability; analytic mixed-norm bound ||A||_2 <= 2.
\[\norm{\A}_2\le2,\qquad \kappa_2(\K_{\rm time})\le\frac{4+\lambda_{\max}^2}{\lambda_{\min}^2}.\]
% source: nonnegative unit-mass kernels split one event's polarity across bins, so ||delta y||_2 <= ||delta y||_1 = 1 (Ch. 2 prop:l1_bump_properties).
{\small One spurious or missed event gives $\norm{\delta\mathbf y}\le1$: the bound limits how far one noise event moves a pixel's history.\par}
\takeaway{Coercivity gives a stable estimate; weakening the penalty weakens these full-space bounds.}
% source detail: Ch. 2, thm:uniqueness, prop:stability.
\src{\dref{Theorem}{thm:uniqueness}, \dref{Proposition}{prop:stability}.}
% source: direct sharper operator bound from Ch. 2 normal equations and Lambda^2 >= lambda_min^2 I.
\talknote{Proof; 90 seconds.}{Walk through the quadratic form. Both terms are nonnegative, and the strictly positive diagonal penalty supplies a lower bound in every direction. Finite-dimensional coercivity gives existence; the positive-definite Hessian gives uniqueness. Subtracting the two normal equations gives the exact perturbation identity.}{The displayed full-space bound uses $\norm{\K^{-1}}_2\le\lambda_{\min}^{-2}$. A sharper operator bound is $\norm{\K^{-1}\A^\top}_2\le1/(2\lambda_{\min})$: from $\A^\top\A\preceq\K-\lambda_{\min}^2I$, the squared singular-value bound is at most $\max_{t\ge\lambda_{\min}^2}(1/t-\lambda_{\min}^2/t^2)=1/(4\lambda_{\min}^2)$. The displayed estimate is a worst-case bound; the reported condition numbers describe the evaluated configurations. Scene terms: one spurious or missed event gives $\norm{\delta\mathbf y}\le1$ (nonnegative unit-mass kernels), so the bound limits how far one noise event moves a history.}
\end{frame}

\begin{frame}{Derive the time-surface constraint by following the edge}
\small
For an ideal single crossing, define
\[\psi(\mathbf x,t)=t-\mathcal T(\mathbf x),\qquad \psi(\mathbf x(t),t)=0.\]
If the edge translates with constant $\dot{\mathbf x}=\vct$, then
\[0=\frac d{dt}\psi(\mathbf x(t),t)=1-\nabla\mathcal T\cdot\vct.\]
{\footnotesize In scene terms: $|\nabla\mathcal T|$ is edge slowness, normal speed $1/|\nabla\mathcal T|$; motion along the edge leaves $\mathcal T$ fixed.\par}
\begin{block}{Model hypotheses}
The edge crosses each location once, has nonzero normal speed, and its recorded timestamp equals a differentiable crossing-time map.
\end{block}
\takeaway{Interpolation, stale timestamps, and repeated crossings make this an approximate data constraint.}
% source detail: Ch. 3, prop:time\_surface; def:time\_surface.
\src{\dref{Proposition}{prop:time_surface}; \dref{Definition}{def:time_surface}.}
\talknote{Geometric derivation; 60 seconds.}{This chain-rule argument is accessible without any sensor-specific machinery. \begin{frame}{Voxel projection: transport a represented event-activity field}
Smooth and bin events into a spatiotemporal field $V(\mathbf x,t)$.
\[a(\mathbf x,s)=\int_s^{s+T}V(\mathbf x,t)\,dt.\]
If $V_t+\vct\cdot\nabla V=0$ with constant velocity over the window,
\[a_s+\vct\cdot\nabla a=0,\qquad a_s=V(\mathbf x,s+T)-V(\mathbf x,s).\]
The voxel-projection residual is $\nabla a\cdot\vct+a_s$.
\takeaway{Activity is a count-derived field. Treating it as transported is an additional modeling assumption.}
% source detail: Ch. 3, subsec:vox\_projection; derivative-operator qualifications.

\begin{frame}{The structure tensor pools directional information}
At a fixed evaluation time $t_\star$, set
\[\mathbf J(\mathbf x)=\int W(\mathbf x-\mathbf y)\,\nabla_3V(\mathbf y,t_\star)\nabla_3V(\mathbf y,t_\star)^\top\,d\mathbf y,
\qquad W\ge0.\]
Use the local residual energy
\[\vct_3^\top\mathbf J\vct_3,\qquad \vct_3=(v_x,v_y,1)^\top,\qquad
\mathbf J=\begin{pmatrix}\widetilde{\mathbf J}&\mathbf j_t\\\mathbf j_t^\top&J_{tt}\end{pmatrix}.\]
\begin{itemize}
\item Each gradient outer product is positive semidefinite and rank at most one.
\item Pooling nonparallel spatial gradients can make $\widetilde{\mathbf J}$ full rank.
\end{itemize}
\takeaway{Pooling can identify both components locally; the global theorem needs only integrated identifiability.}
% source detail: Ch. 3, subsec:vox\_struct, prop:vox\_tensor\_wellposed.
\src{\dref{Section}{subsec:vox_struct}, \dref{Proposition}{prop:vox_tensor_wellposed}.}
\talknote{Structure tensor; 75 seconds.}{Expand the quadratic: v\textasciicircum{}T Jtilde v + 2 j\_t\textasciicircum{}T v + J\_tt. The last term is independent of velocity, so the spatial block is the data curvature and the temporal cross terms give the forcing. The temporal component of v\_3 is fixed to one; this is an ordinary least-squares transport model, not a total-least-squares eigenvector model.}{Nonnegative pooling preserves positive semidefiniteness, since z\textasciicircum{}T J z is an integral of nonnegative squared directional derivatives. A locally planar event surface is a model assumption, and smoothing scale limits temporal detail. The theorem will not require every pixel's spatial block to be positive definite.}
\end{frame}

\begin{frame}{The main flow theorem: global information controls constants}
\begin{block}{Unified well-posedness theorem}
Let $\Omega\subset\R^2$ be bounded, connected, and Lipschitz; $\alpha>0$.
Assume
\[\B\in L^\infty,\quad \B=\B^\top\succeq0\text{ a.e.},\quad b\in L^2,\qquad
\int_\Omega\B(\mathbf x)\,d\mathbf x\succ0.\]
Then $E_{\B,b}$ has a unique minimizer in $H^1(\Omega;\R^2)$.
\end{block}
\vspace{1mm}
The integrated matrix must identify \textbf{every constant velocity direction}.\par
\small In scene terms: edges in at least two orientations somewhere in the image; neighbors supply the motion where $\B=0$.
\takeaway{$\B$ may vanish on open regions; integrated directional information and smoothness still give a unique estimate.}
% source detail: Ch. 3, thm:unified\_wellposedness, eq:integrated\_identifiability.
\src{\dref{Theorem}{thm:unified_wellposedness}, \dref{Eq.}{eq:integrated_identifiability}.}
\talknote{Central theorem; 90 seconds.}{Classical quadratic optical flow already has well-posedness theory, including Weickert and Schn\"orr. The contribution here is one analysis for three event-derived coefficient constructions, with explicit identifiability and perturbation hypotheses. Read the hypotheses and interpret each. Connectedness makes the kernel of the gradient consist of global constants. The positive regularization coefficient controls spatial variation. Bounded data curvature and L2 forcing make the weak forms bounded. Positive definiteness of the integrated B controls constant fields that smoothness cannot see.}{B may vanish on open subsets while its integral remains positive definite. \begin{frame}{Proof: control the mean and the fluctuation separately}
Decompose $\vct=\underbrace{\bar{\vct}}_{\text{common drift (pan)}}+\underbrace{\mathbf w}_{\text{variation across the image}}$, with $\int_\Omega\mathbf w=0$.
\[\norm{\mathbf w}_{L^2}\le C_P\norm{\nabla\vct}_{L^2}.\]
Let $\beta_B=\norm{\B}_{L^\infty}$ and $m=\lambda_{\min}(\int_\Omega\B)>0$ (how well the edges cover every direction).
% source: Ch. 3 proof of thm:unified_wellposedness, eq:mean_control.
\[m|\bar{\vct}|^2\le2\int_\Omega\vct^\top\B\vct+2\beta_BC_P^2\norm{\nabla\vct}_{L^2}^2.\]
Therefore the bilinear form satisfies
% source: Ch. 3 proof of thm:unified_wellposedness; admissible lower bound, not optimal.
\[\mathcal B(\vct,\vct)=\int\vct^\top\B\vct+\alpha^2\int|\nabla\vct|^2
\ge c_0\norm{\vct}_{H^1}^2,\qquad
c_0=c_0(\Omega,\alpha,\beta_B,m).\]
\takeaway{Poincar\'e controls fluctuations, data control the mean, and Lax--Milgram supplies the unique weak solution.}
% source detail: Ch. 3, proof of thm:unified\_wellposedness.
\src{proof of \dref{Theorem}{thm:unified_wellposedness}.}
\talknote{Main proof; 2 minutes.}{The gradient seminorm misses exactly the constant vector fields. Write the field as its mean plus a zero-mean part. Poincar\'e controls the latter. Use $\norm{\B^{1/2}\bar{\vct}}_{L^2}\le\norm{\B^{1/2}\vct}_{L^2}+\norm{\B^{1/2}\mathbf w}_{L^2}$, square with the factor-two inequality, and bound the fluctuation term by $\beta_BC_P^2\norm{\nabla\vct}_{L^2}^2$. Positive definiteness of the integrated matrix controls the mean.}{An admissible lower bound, not an optimal constant, is $c_0=\min\{m/(2|\Omega|),\alpha^2/(1+C_P^2+2|\Omega|\beta_BC_P^2/m)\}$. Insert it into $\norm{\vct}_{H^1}^2=|\Omega||\bar{\vct}|^2+\norm{\mathbf w}_{L^2}^2+\norm{\nabla\vct}_{L^2}^2$. Lax--Milgram gives the unique minimizer. For fixed positive $\beta_B,\alpha$, this lower bound deteriorates linearly as $m$ decreases. Scene terms: mean = common drift (camera pan), fluctuation = variation across the image; smoothness charges nothing for a pan, so the edges must fix it.}
\end{frame}

\begin{frame}{Stability follows by subtracting weak equations}
Suppose two coefficient pairs share bounds
\[\norm{\B_i}_{L^\infty}\le \beta_B,\qquad\lambda_{\min}\!\left(\int\B_i\right)\ge m>0,
\qquad\norm{b_i}_{L^2}\le R_b.\]
Here $c_0=c_0(\Omega,\alpha,\beta_B,m)$ is independent of $b_i$.
For $\mathbf z=\vct_1^*-\vct_2^*$,
\[\mathcal B_1(\mathbf z,\mathbf z)
=\int(b_1-b_2)\cdot\mathbf z+\int\mathbf z^\top(\B_2-\B_1)\vct_2^*.\]
Hence
\[\norm{\mathbf z}_{H^1}\le c_0^{-1}\norm{b_1-b_2}_{L^2}
+c_0^{-2}R_b\norm{\B_1-\B_2}_{L^\infty}.\]
{\footnotesize In scene terms: coefficient changes come from timestamp jitter, binning, and noise events; the constants grow as $m$ shrinks (stripes, slide 29).\par}
\takeaway{The estimator is Lipschitz in its coefficients while identifiability remains uniformly positive.}
% source detail: Ch. 3, thm:unified\_wellposedness, eq:abstract\_stability.
\src{\dref{Theorem}{thm:unified_wellposedness}, \dref{Eq.}{eq:abstract_stability}.}
\talknote{Stability proof; 75 seconds.}{The same coercivity constant works on both coefficient pairs because the quantitative bounds are shared. Subtract their weak equations and test with the solution difference. Cauchy--Schwarz controls the right side, and $\norm{\vct_2^*}_{H^1}\le R_b/c_0$ follows from its own weak equation. Divide by $\norm{\mathbf z}_{H^1}$ if it is nonzero.}{The conclusion is stability of the chosen estimator, not an error bound to ground-truth velocity. The estimate may become poor as $m\to0$. If a density-dependent diffusion tensor is recomputed, the regularizer also changes; its $L^\infty$ perturbation must enter the bound. Scene terms: coefficient changes come from timestamp jitter, binning, and noise events; small $m$ is the stripe case on slide 29.}
\end{frame}

\begin{frame}{Discrete flow: connectivity and sampled directions matter}
\small
Fix $\alpha>0$, $q_i>0$, and $\B_i=\B_i^\top\succeq0$ on a connected graph with
$w_{ij}=w_{ji}\ge0$ and degree $d_i=\sum_jw_{ij}>0$.
\begin{block}{One simultaneous block-Jacobi sweep}
% source: Ch. 3 thm:convergence and cor:convergence_psd_blocks; normal-equation block update with positive q_i absorbed into B_i and b_i.
\[
\underbrace{(q_i\B_i+\alpha^2d_i\mathbf I_2)}_{\text{local }2\times2\text{ matrix}}
\vct_i^{(k+1)}
=q_i b_i+\alpha^2\sum_jw_{ij}\vct_j^{(k)}.
\]
\end{block}
\begin{itemize}
\item At each node, solve for two velocity components using local data and the previous neighbor velocities.
\item Update every node from the same old field; then repeat the sweep.
\item Scene terms: each pixel blends its edge measurement with neighbors' motion.
\end{itemize}
% source: Ch. 3 thm:convergence and cor:convergence_psd_blocks, extended to q_i>0 by absorbing q_i into B_i.
If $\sum_iq_i\B_i\succ0$, the discrete energy has a unique minimizer, and these sweeps converge to it from every initial field.
% source detail: Ch. 3, cor:vox\_projection\_wellposed; positive-$q_i$ extension of thm:convergence and cor:convergence\_psd\_blocks.
\src{\dref{Theorem}{thm:convergence} and \dref{Corollary}{cor:convergence_psd_blocks}; extension to positive node factors $q_i$.}
\talknote{Numerical construction and theorem; about 90 seconds.}{Each node stores a two-component velocity. With coefficients fixed, the normal equation at node i has the local data matrix $q_i\B_i$ plus alpha squared times the weighted degree. Move the neighboring velocities to the right, and use their values from the previous sweep. This gives the displayed two-by-two solve. The $q_i b_i$ term carries the local observations, and the weighted neighbor sum carries the previous surrounding field. Compute every new velocity from the same old field, then replace the whole field and repeat.}{The theorem assumes a connected graph, symmetric nonnegative edge weights, positive degrees and node factors, PSD data blocks, and a positive-definite sampled Gram matrix. For this graph operator, zero graph energy forces a constant field, and the sampled data remove that constant null direction. Write $H=D-R$, where $D$ contains the local two-by-two blocks and $R$ contains the neighbor couplings. Both $D-R$ and $D+R$ are positive definite: possible alternating constants of the signless graph form are removed by the same sampled data condition. Hence the spectrum of the similar symmetric Jacobi matrix lies in $(-1,1)$. This proves convergence of the simultaneous schedule from every initial field. Match a representation's stencil and update schedule to these hypotheses before applying the result. Scene terms: information spreads one neighbor per sweep.}
\end{frame}

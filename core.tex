\section{Temporal reconstruction}
\begin{frame}{Start with one pixel: integrated changes on a temporal chain}
\begin{columns}[T]
\column{.54\textwidth}
Choose a unit-mass temporal kernel $\phi_{s_k}$:
\[e(t)=\sum_{k\text{ at this pixel}}p_k\phi_{s_k}(t).\]
\[y_i=\int_{t_i}^{t_{i+1}}e(t)\,dt.\]
The reconstruction times $t_i$ are chosen by the algorithm.
\column{.43\textwidth}
\[\A=\begin{pmatrix}-1&1&&\\&-1&1&\\&&\ddots&\ddots\end{pmatrix}.\]
\[\A\mathbf f\approx\mathbf y.\]
\[\ker\A=\operatorname{span}\{\mathbf1\}.\]
Events determine increments, leaving a constant free.
\end{columns}
\takeaway{Kernel width controls temporal averaging; a baseline requires an additional choice.}
\src{Ch. 2, eq:exy, eq:Exy, eq:tempdiffA, lem:nullspace.}
\talknote{Temporal observation model; 75 seconds.}{Suppress the pixel index to keep the notation readable. Integrating a unit-mass kernel across a bin assigns all or part of each event to that bin. Delta binning has no spread; Gaussian, rectangular, and spline kernels distribute influence over time. The entries of A are unscaled increments, not derivatives divided by a time step.}{If Af=0, every consecutive pair of components is equal, hence f is constant. Conversely A1=0. Under ideal threshold calibration the observed increments approximate changes in L/c\_th, with kernel-boundary, overshoot, and between-event endpoint errors. No absolute frame is used to set the baseline.}
\end{frame}

\begin{frame}{A regularized estimate is uniquely selected}
For fixed $\Lam=\operatorname{diag}(\lambda_i)$ with every $\lambda_i>0$,
\[\mathbf f^*=\arg\min_{\mathbf f\in\R^n}\left\{\frac12\norm{\A\mathbf f-\mathbf y}^2+\frac12\norm{\Lam\mathbf f}^2\right\}.\]
\begin{block}{Temporal uniqueness theorem}
\[\K_{\rm time}=\A^\top\A+\Lam^\top\Lam\succ0,\qquad \K_{\rm time}\mathbf f^*=\A^\top\mathbf y.\]
The fixed-weight problem has a unique global minimizer.
\end{block}
% source: Direct consequence of Ch. 2 thm:uniqueness normal equations and A1=0; supplementary identity.
The selected baseline satisfies $\mathbf1^\top\Lam^2\mathbf f^*=0$.
\takeaway{The penalty chooses an unobserved baseline and also changes observable increments.}
\src{Ch. 2, eq:in, thm:uniqueness. Baseline identity: direct consequence.}
\talknote{Model and theorem; 75 seconds.}{The objective balances fidelity to observed changes against weighted magnitude. The normal equations follow by differentiating the quadratic. Solve this system by factorization rather than explicitly forming an inverse. Pixels are independent, so these temporal solves can be batched.}{Premultiply the normal equations by $\mathbf1^\top$ and use $\A\mathbf1=0$ to obtain the weighted zero-mean identity. Under the fixed model $\mathbf y\mid\mathbf f\sim N(\A\mathbf f,I)$ and $\mathbf f\sim N(0,(\Lam^\top\Lam)^{-1})$, this minimizer is the Gaussian MAP estimate. The unit noise covariance and fixed positive $\Lam$ are assumptions; data-derived weights give a plug-in fixed-weight interpretation, not a full-pipeline MAP or stability theorem. As scalar $\lambda\to0$, the solution tends to the minimum-norm exact fit, which sets the constant null mode to zero.}
\end{frame}

\begin{frame}{Why uniqueness and stability hold}
Write $\lambda_{\min}=\min_i\lambda_i$. For $\mathbf z\ne0$,
\[\mathbf z^\top\K_{\rm time}\mathbf z=\norm{\A\mathbf z}^2+\norm{\Lam\mathbf z}^2\ge\lambda_{\min}^2\norm{\mathbf z}^2>0.\]
At the same fixed weights,
\[\delta\mathbf f^*=\K_{\rm time}^{-1}\A^\top\delta\mathbf y,\qquad
\norm{\delta\mathbf f^*}\le\frac{\norm{\A}_2}{\lambda_{\min}^2}\norm{\delta\mathbf y}.\]
% source: Ch. 2 thm:uniqueness and prop:stability; analytic mixed-norm bound ||A||_2 <= 2.
\[\norm{\A}_2\le2,\qquad \kappa_2(\K_{\rm time})\le\frac{4+\lambda_{\max}^2}{\lambda_{\min}^2}.\]
\takeaway{Coercivity gives a stable estimate; weakening the penalty weakens these full-space bounds.}
\src{Ch. 2, thm:uniqueness, prop:stability.}
% source: direct sharper operator bound from Ch. 2 normal equations and Lambda^2 >= lambda_min^2 I.
\talknote{Proof; 90 seconds.}{Walk through the quadratic form. Both terms are nonnegative, and the strictly positive diagonal penalty supplies a lower bound in every direction. Finite-dimensional coercivity gives existence; the positive-definite Hessian gives uniqueness. Subtracting the two normal equations gives the exact perturbation identity.}{The displayed full-space bound uses $\norm{\K^{-1}}_2\le\lambda_{\min}^{-2}$. A sharper operator bound is $\norm{\K^{-1}\A^\top}_2\le1/(2\lambda_{\min})$: from $\A^\top\A\preceq\K-\lambda_{\min}^2I$, the squared singular-value bound is at most $\max_{t\ge\lambda_{\min}^2}(1/t-\lambda_{\min}^2/t^2)=1/(4\lambda_{\min}^2)$. The original estimate is a worst-case bound; reported CN is a measured system condition number.}
\end{frame}

\begin{frame}{Shared motion couples temporal profiles to one scene}
% source: Ch. 2 eq:ch2_sar_projection, eq:ch2_sar_nullspace, prop:ch2_sar_solvability; fixed row-normalized projections.
For $n$ frames, stack the fixed, row-normalized known-motion footprint projections $P_k$ into $P$; $P_k\mathbf1=\mathbf1$.
With $A\in\R^{(n-1)\times n}$ the first-difference matrix,
\[B_{\rm SAR}=(A\otimes I)P,\qquad
\ker B_{\rm SAR}=\{\theta:P_{k+1}\theta=P_k\theta,\ 1\le k<n\},\qquad
B_{\rm SAR}\mathbf1=0.\]
With an unsquared coefficient $\lambda>0$,
\[\left(B_{\rm SAR}^\top B_{\rm SAR}+\lambda I\right)\widehat\theta
=B_{\rm SAR}^\top b.\]
\takeaway{Motion links per-frame changes to a shared scene while retaining motion-invariant scene modes.}
\src{Ch. 2, prop:ch2\_sar\_solvability.}
\talknote{Shared-scene temporal model; 75 seconds.}{Here $\theta$ contains static scene values only. The projections $P_k$ encode fixed, known motion and map that scene to each frame; $A$ differences the profiles. The nullspace contains parameters whose projected profiles agree across frames. Constants are included because each projection is row-normalized; other invariant scene modes may also occur. This resembles flow's aperture ambiguity in that observations leave directions uncontrolled, but proves no motion ambiguity for SAR.}{Here lambda is the unsquared ridge coefficient in the normal equations. For this fixed-motion linear model, the stability estimate is $\norm{\delta\theta}\le\norm{B_{\rm SAR}}\norm{\delta b}/\lambda$. \begin{frame}{Motion introduces a new invisible direction}
\begin{columns}[c]
\column{.49\textwidth}

\end{tikzpicture}
\column{.48\textwidth}
A time surface $\mathcal T(\mathbf x)$ records an edge's crossing time.
\[\nabla\mathcal T\cdot\vct=1.\]
One scalar equation constrains a vector velocity.
\[\vct=\frac{\nabla\mathcal T}{|\nabla\mathcal T|^2}+\tau\nabla\mathcal T^\perp.\]
\end{columns}
\takeaway{This is the aperture problem: an edge does not reveal motion along its tangent.}
\src{Ch. 3, prop:time\_surface and following aperture remark.}
\talknote{Transition and aperture problem; 75 seconds.}{Define optical flow as apparent image-plane velocity, rather than a physical three-dimensional velocity. The crossing-time surface has units of time, so its spatial gradient has units time/pixel and its dot product with pixels/time is dimensionless. This also makes the right-hand side one intuitive.}{The formula assumes a nonzero gradient. For a single translating straight edge, changing tangential velocity produces the same crossing-time observations. Regularizing spatial derivatives alone also leaves constant tangential velocities free unless the scene supplies another identifying direction.}
\end{frame}

\begin{frame}{Derive the time-surface constraint by following the edge}
\small
For an ideal single crossing, define
\[\psi(\mathbf x,t)=t-\mathcal T(\mathbf x),\qquad \psi(\mathbf x(t),t)=0.\]
If the edge translates with constant $\dot{\mathbf x}=\vct$, then
\[0=\frac d{dt}\psi(\mathbf x(t),t)=1-\nabla\mathcal T\cdot\vct.\]
\begin{block}{Model hypotheses}
The edge crosses each location once, has nonzero normal speed, and its recorded timestamp equals a differentiable crossing-time map.
\end{block}
\takeaway{Interpolation, stale timestamps, and repeated crossings make this an approximate data constraint.}
\src{Ch. 3, prop:time\_surface; def:time\_surface.}
\talknote{Geometric derivation; 60 seconds.}{This chain-rule argument is accessible without any sensor-specific machinery. Follow a material point on the translating edge and differentiate the identity saying it stays on the zero level set.}{The coefficient bounds used later apply to fitted or mollified maps. The timestamp-regression result assumes a differentiable crossing-time map with Lipschitz Hessian, but raw last-timestamp maps do not automatically satisfy those assumptions. A timestamp may belong to an old edge or multiple crossings; the variational theorem bounds usable coefficient perturbations under uniform identifiability.}
\end{frame}

\begin{frame}{Time-surface gradient estimation: bias versus variance}
\small
Fit a plane to timestamps at offsets $\mathbf z_j$ around a pixel.
\[Q=\sum_j\mathbf z_j\mathbf z_j^\top\succ0.\]
For a centrally symmetric design and independent timestamp errors with common mean and finite variance $\sigma^2$:
\[\operatorname{Cov}(\widehat{\nabla\mathcal T})=\sigma^2 Q^{-1}.\]
With a Lipschitz Hessian of constant $L_H$ on a neighborhood containing the sample disk,
% source: Ch. 3 prop:ts_regression_bound(iv), eq:ts_discrete_bias and eq:ts_discrete_mse; L_H renames its Lipschitz constant L.
\[\norm{\operatorname{Bias}}\le\frac{L_H}{6}\norm{Q^{-1}}\sum_j\norm{\mathbf z_j}^4,\qquad
\operatorname{MSE}=\norm{\operatorname{Bias}}^2+\sigma^2\operatorname{tr}(Q^{-1}).\]
\takeaway{A larger neighborhood reduces variance but can increase curvature bias. Regression rank and velocity identifiability are different questions.}
\src{Ch. 3, prop:ts\_regression\_bound.}
\talknote{Gradient-estimation result; 75 seconds; shorten to the takeaway in a compressed talk.}{The design matrix Q depends on sample positions, not the observed slope. Central symmetry decouples the intercept and cancels the quadratic Taylor term in the slope. A Lipschitz Hessian leaves a cubic remainder bounded by L\_H |z|\textasciicircum{}3/6. Multiplication by z and Q\textasciicircum{}{-1} yields the bias bound. Independent equal-variance errors give the covariance, and the MSE decomposition combines both contributions for the same discrete estimator.}{A full-rank disk of timestamps around a moving straight edge can estimate its gradient well while the tangential velocity remains unidentified. The assumed timestamp-noise model is not derived from event-count statistics. The asymptotic radius heuristic is in the backup, with its finite-design qualifications.}
\end{frame}

\begin{frame}{Voxel projection: transport a represented event-activity field}
Smooth and bin events into a spatiotemporal field $V(\mathbf x,t)$.
\[a(\mathbf x,s)=\int_s^{s+T}V(\mathbf x,t)\,dt.\]
If $V_t+\vct\cdot\nabla V=0$ with constant velocity over the window,
\[a_s+\vct\cdot\nabla a=0,\qquad a_s=V(\mathbf x,s+T)-V(\mathbf x,s).\]
The voxel-projection residual is $\nabla a\cdot\vct+a_s$.
\takeaway{Activity is a count-derived field. Treating it as transported is an additional modeling assumption.}
\src{Ch. 3, subsec:vox\_projection; derivative-operator qualifications.}
\talknote{Voxel model; 75 seconds.}{A voxel grid adds a time axis to the image plane and stores represented event contributions in time bins. The activity field is not an intensity image: a bright stationary scene can have zero activity. For an exactly transported density with constant velocity, integrate the PDE over the sliding window; this gives the displayed transport identity exactly, even for a large displacement.}{Finite temporal projection changes the profile by averaging its translates. The evaluated spatial bin mean and temporal OLS slope need not be derivative-consistent partners, and asymmetric smoothing can add mismatch. The well-posedness theorem only needs fixed bounded coefficient fields; an accurate physical-velocity interpretation also needs the transport approximation.}
\end{frame}

\begin{frame}{The structure tensor pools directional information}
At a fixed evaluation time $t_\star$, set
\[\mathbf J(\mathbf x)=\int W(\mathbf x-\mathbf y)\,\nabla_3V(\mathbf y,t_\star)\nabla_3V(\mathbf y,t_\star)^\top\,d\mathbf y,
\qquad W\ge0.\]
Use the local residual energy
\[\vct_3^\top\mathbf J\vct_3,\qquad \vct_3=(v_x,v_y,1)^\top,\qquad
\mathbf J=\begin{pmatrix}\widetilde{\mathbf J}&\mathbf j_t\\\mathbf j_t^\top&J_{tt}\end{pmatrix}.\]
\begin{itemize}
\item Each gradient outer product is positive semidefinite and rank at most one.
\item Pooling nonparallel spatial gradients can make $\widetilde{\mathbf J}$ full rank.
\end{itemize}
\takeaway{Pooling can identify both components locally; the global theorem needs only integrated identifiability.}
\src{Ch. 3, subsec:vox\_struct, prop:vox\_tensor\_wellposed.}
\talknote{Structure tensor; 75 seconds.}{Expand the quadratic: v\textasciicircum{}T Jtilde v + 2 j\_t\textasciicircum{}T v + J\_tt. The last term is independent of velocity, so the spatial block is the data curvature and the temporal cross terms give the forcing. The temporal component of v\_3 is fixed to one; this is an ordinary least-squares transport model, not a total-least-squares eigenvector model.}{Nonnegative pooling preserves positive semidefiniteness, since z\textasciicircum{}T J z is an integral of nonnegative squared directional derivatives. A locally planar event surface is a model assumption, and smoothing scale limits temporal detail. The theorem will not require every pixel's spatial block to be positive definite.}
\end{frame}

Rename the flow data matrix $\B$ to distinguish it from temporal $\A$:
\[E_{\B,b}(\vct)=\frac12\int_\Omega\bigl(\vct^\top\B\vct-2b^\top\vct\bigr)
+\frac{\alpha^2}{2}\int_\Omega|\nabla\vct|^2.\]
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{lll}\toprule Representation & $\B$ & $b$\\\midrule
Time surface & $\nabla\mathcal T\nabla\mathcal T^\top$ & $\nabla\mathcal T$\\
Voxel projection & $\nabla a\nabla a^\top$ & $-a_s\nabla a$\\

\end{tabular}
\takeaway{The representations change the coefficients; the analysis can be proved once.}
\src{Ch. 3, thm:unified\_wellposedness and its three corollaries.}
\talknote{Unification; 60 seconds.}{The energies agree with the representation-specific residual energies up to an overall positive factor and a constant independent of flow. The common space is H\textasciicircum{}1(Omega; R\textasciicircum{}2), and the Dirichlet term couples neighboring velocities.}{The notation B here is A\_data in the dissertation, and alpha is alpha\_smooth. The change avoids using A for two different operators on consecutive chapters. The common analysis is about fixed coefficients. A good mathematical estimator can still encode a poor representation of physical motion.}
\end{frame}

\begin{frame}{The main flow theorem: global information controls constants}
\begin{block}{Unified well-posedness theorem}
Let $\Omega\subset\R^2$ be bounded, connected, and Lipschitz; $\alpha>0$.
Assume
\[\B\in L^\infty,\quad \B=\B^\top\succeq0\text{ a.e.},\quad b\in L^2,\qquad
\int_\Omega\B(\mathbf x)\,d\mathbf x\succ0.\]
Then $E_{\B,b}$ has a unique minimizer in $H^1(\Omega;\R^2)$.
\end{block}
\vspace{2mm}
The integrated matrix must identify \textbf{every constant velocity direction}.
\takeaway{$\B$ may vanish on open regions; integrated directional information and smoothness still give a unique estimate.}
\src{Ch. 3, thm:unified\_wellposedness, eq:integrated\_identifiability.}
\talknote{Central theorem; 90 seconds.}{Read the hypotheses and interpret each. Connectedness makes the kernel of the gradient consist of global constants. The positive regularization coefficient controls spatial variation. Bounded data curvature and L2 forcing make the weak forms bounded. Positive definiteness of the integrated B controls constant fields that smoothness cannot see.}{B may vanish on open subsets while its integral remains positive definite. \begin{frame}{Hard fusion changes both the constraint and the regularization}
Joint intensity fusion (JIF) imposes $\fL=\fR=\mathbf v$ and solves
\[(2\A^\top\W\A+\Lam^2)\mathbf v=\A^\top\W(\yL+\yR).\]
For fixed equal weights, the unchanged soft model has the limit
\[\gamma\to\infty:\qquad\fL,\fR\to(\A^\top\W\A+\Lam^2)^{-1}\frac12\A^\top\W(\yL+\yR).\]
